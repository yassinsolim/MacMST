cmake_minimum_required(VERSION 3.20)
project(MacMST VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MACMST_ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(MACMST_ENABLE_HARDWARE_TESTS "Register opt-in public read-only macOS hardware tests" OFF)

add_library(macmst_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(macmst_options INTERFACE
        -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
    if(MACMST_ENABLE_SANITIZERS)
        target_compile_options(macmst_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(macmst_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

include(CTest)
add_library(macmst_displayport src/displayport/dpcd/capabilities.cpp)
target_include_directories(macmst_displayport PUBLIC src)
target_link_libraries(macmst_displayport PUBLIC macmst_options)
if(BUILD_TESTING)
    add_executable(macmst_dpcd_tests tests/dpcd_tests.cpp)
    target_link_libraries(macmst_dpcd_tests PRIVATE macmst_displayport)
    add_test(NAME dpcd_capabilities COMMAND macmst_dpcd_tests)
    set_tests_properties(dpcd_capabilities PROPERTIES LABELS unit)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    set(CMAKE_OBJCXX_EXTENSIONS OFF)
    find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
    find_library(COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
    find_library(IOKIT_FRAMEWORK IOKit REQUIRED)

    add_library(macmst_macos src/platform/macos/probe.mm)
    target_include_directories(macmst_macos PUBLIC src)
    target_compile_options(macmst_macos PRIVATE -fobjc-arc)
    target_link_libraries(macmst_macos PUBLIC macmst_options
        ${FOUNDATION_FRAMEWORK} ${COREGRAPHICS_FRAMEWORK} ${IOKIT_FRAMEWORK})

    add_executable(macmst src/cli/main.cpp)
    target_link_libraries(macmst PRIVATE macmst_macos)
    if(BUILD_TESTING)
        add_executable(macmst_probe_tests tests/probe_tests.mm)
        target_compile_options(macmst_probe_tests PRIVATE -fobjc-arc)
        target_link_libraries(macmst_probe_tests PRIVATE macmst_macos)
        add_test(NAME probe_privacy COMMAND macmst_probe_tests)
        set_tests_properties(probe_privacy PROPERTIES LABELS unit)
        add_test(NAME cli_help COMMAND macmst --help)
        add_test(NAME cli_reject_unknown COMMAND macmst enable-mst)
        set_tests_properties(cli_reject_unknown PROPERTIES WILL_FAIL TRUE)
        set_tests_properties(cli_help cli_reject_unknown PROPERTIES LABELS unit)
    endif()
endif()

if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME capture_privacy
        COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_capture_baseline.py -v)
    set_tests_properties(capture_privacy PROPERTIES WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} LABELS unit)
    add_test(NAME iodp_static_parsers
        COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_iodp_static.py -v)
    set_tests_properties(iodp_static_parsers PROPERTIES WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} LABELS unit)
    if(APPLE)
        add_test(NAME cli_contract
            COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_probe_cli.py $<TARGET_FILE:macmst>)
        set_tests_properties(cli_contract PROPERTIES LABELS unit)
        if(MACMST_ENABLE_HARDWARE_TESTS)
            add_test(NAME probe_hardware
                COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_probe_cli.py $<TARGET_FILE:macmst> --hardware)
            set_tests_properties(probe_hardware PROPERTIES LABELS hardware)
        endif()
    endif()
endif()
